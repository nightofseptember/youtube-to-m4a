@echo off
title yt-dlp M4A Downloader

echo ==============================
echo   YouTube to M4A Downloader
echo ==============================
echo.
echo Paste YouTube links (one per line)
echo press ENTER to start a new line (to put new links in)
echo When finished putting in links, press CTRL+Z then enter.
echo.

yt-dlp ^
-f bestaudio[ext=m4a] ^
--embed-thumbnail ^
--add-metadata ^
-o "%%(title)s - %%(artist,creator,uploader)s.%%(ext)s" ^
-a -

echo.
echo Download finished.
pause
