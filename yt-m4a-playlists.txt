@echo off
title yt-dlp M4A Downloader

echo ==============================
echo   YouTube to M4A Downloader
echo   (Videos ^& Playlists)
echo ==============================
echo.
set /p URL=Paste YT playlist URL and press ENTER:
echo.

"%~dp0ytfiles\yt-dlp_win_x86\yt-dlp_x86.exe" ^
 --ffmpeg-location "%~dp0ytfiles\ffmpeg-master-latest-win64-gpl-shared\ffmpeg-master-latest-win64-gpl-shared\bin" ^
 -f bestaudio[ext=m4a] ^
 --embed-thumbnail ^
 --add-metadata ^
 --yes-playlist ^
 -o "%%(playlist_title)s\%%(playlist_index)s - %%(title)s.%%(ext)s" ^
 "%URL%"

echo.
echo Download finished.
pause
